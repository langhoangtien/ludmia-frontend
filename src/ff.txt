limit_req_zone $binary_remote_addr zone=global_zone:20m rate=30r/s;
limit_req_zone $binary_remote_addr zone=client_zone:10m rate=10r/s;
limit_req_zone $binary_remote_addr zone=payment_zone:10m rate=5r/s;

server {
    server_name api.naturaeon.com;

 

    location /client/ {
        # Giới hạn chặt hơn cho các route POST từ client
        limit_req zone=client_zone burst=5 nodelay;
        proxy_pass http://localhost:4000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }

    location /payment/ {
        # Giới hạn chặt hơn cho các route POST từ client
        limit_req zone=payment_zone burst=2 nodelay;
        proxy_pass http://localhost:4000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }


    location / {
        proxy_pass http://localhost:4000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/api.naturaeon.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/api.naturaeon.com/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}
server {
    if ($host = api.naturaeon.com) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    server_name api.naturaeon.com;
    return 404; # managed by Certbot


}
